declare
  v_total integer;
  v_gaps integer;
  v_alias_linked integer;
  v_unclassified integer;
begin
  select count(*),
         count(*) filter(where source_mapping_identity is null),
         count(*) filter(where raw_summary->>'legacy_path_alias_resolution'='confirmed'),
         count(*) filter(where classification_status='unclassified')
  into v_total,v_gaps,v_alias_linked,v_unclassified
  from fitmatch_catalog.source_product_snapshots
  where run_id='a141f6bf-5218-4c87-9ce2-e77557472f37'::uuid;

  if v_total<>383 or v_gaps<>0 or v_alias_linked<>80 or v_unclassified<>14 then

      v_total,v_gaps,v_alias_linked,v_unclassified;
  end if;
end
